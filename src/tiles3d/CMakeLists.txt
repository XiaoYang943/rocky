# Cesium Native and its GLM definitions stay private to the Tiles3D DLL.
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF)
set(CESIUM_USE_EZVCPKG OFF CACHE BOOL "" FORCE)
set(CESIUM_TESTS_ENABLED OFF CACHE BOOL "" FORCE)
set(CESIUM_ENABLE_CLANG_TIDY OFF CACHE BOOL "" FORCE)
add_subdirectory("${GN_SOURCE_cesium-native}" cesium-native EXCLUDE_FROM_ALL)
target_compile_options(CesiumCurl PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNOMINMAX>)
find_package(Microsoft.GSL CONFIG REQUIRED)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
set(VSGCS_USE_PROJ OFF)
set(VSGCS_FULL_DATA_DIR "${CMAKE_INSTALL_PREFIX}/share/GeoStudioNative/resources/tiles3d")
set(vsgCs_VERSION "0.2.0")
set(vsgCs_VERSION_MAJOR 0)
set(vsgCs_VERSION_MINOR 2)
set(vsgCs_VERSION_PATCH 0)
set(vsgCs_VERSION_TWEAK 0)
set(TILES3D_PRIVATE_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/include")
configure_file(runtime/Config.h.in "${TILES3D_PRIVATE_INCLUDE}/geostudio/native/tiles3d/Config.h")
configure_file(runtime/Version.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/Version.cpp")
include(CheckIncludeFileCXX)
check_include_file_cxx("sys/utsname.h" HAVE_SYS_UTSNAME)
file(GLOB SOURCES CONFIGURE_DEPENDS runtime/*.cpp)
list(REMOVE_ITEM SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/runtime/CppAllocator.cpp")
add_library(geostudio-tiles3d-runtime STATIC ${SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/Version.cpp")
set_target_properties(geostudio-tiles3d-runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(geostudio-tiles3d-runtime PUBLIC cxx_std_20)
target_compile_definitions(geostudio-tiles3d-runtime PUBLIC VSGCS_LIBRARY_STATIC VSGCS_STATIC_DEFINE PRIVATE $<$<BOOL:${HAVE_SYS_UTSNAME}>:HAVE_SYS_UTSNAME>)
target_compile_options(geostudio-tiles3d-runtime PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/bigobj>)
if(WIN32)
    target_compile_definitions(geostudio-tiles3d-runtime PRIVATE NOGDI NOMINMAX)
endif()
target_include_directories(geostudio-tiles3d-runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" "${TILES3D_PRIVATE_INCLUDE}")
target_link_libraries(geostudio-tiles3d-runtime PUBLIC vsg::vsg Cesium3DTilesSelection CesiumRasterOverlays CesiumGltfContent CesiumGltfReader CesiumGltf CesiumCurl Microsoft.GSL::GSL spdlog::spdlog
    PRIVATE CURL::libcurl OpenSSL::SSL)
include(GenerateExportHeader)
generate_export_header(geostudio-tiles3d-runtime BASE_NAME VSGCS EXPORT_FILE_NAME "${TILES3D_PRIVATE_INCLUDE}/geostudio/native/tiles3d/Export.h")
add_library(geostudio-tiles3d SHARED TilesetAdapter.cpp)
set_target_properties(geostudio-tiles3d PROPERTIES EXPORT_NAME Tiles3D)
target_compile_features(geostudio-tiles3d PRIVATE cxx_std_20)
target_compile_definitions(geostudio-tiles3d PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(geostudio-tiles3d PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8> $<$<CXX_COMPILER_ID:MSVC>:/bigobj>)
target_link_libraries(geostudio-tiles3d PUBLIC vsg::vsg nlohmann_json::nlohmann_json PRIVATE geostudio-tiles3d-runtime)
target_include_directories(geostudio-tiles3d PUBLIC
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${GEOSTUDIO_GENERATED_INCLUDE}>
    $<INSTALL_INTERFACE:include>)
generate_export_header(geostudio-tiles3d BASE_NAME GEOSTUDIO_NATIVE EXPORT_FILE_NAME "${GEOSTUDIO_GENERATED_INCLUDE}/geostudio/native/Export.h")
install(TARGETS geostudio-tiles3d EXPORT GeoStudioNativeTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
