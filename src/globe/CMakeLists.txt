# Retain upstream feature guards while exposing the GeoStudio API and ABI.
set(PROJECT_VERSION_ABI 61)
find_package(PROJ CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)
find_package(GDAL REQUIRED)
find_package(CURL CONFIG REQUIRED)
find_package(EnTT CONFIG REQUIRED)
find_package(Vulkan 1.3.261 REQUIRED)
set(ROCKY_MAX_NUMBER_OF_VIEWS 8 CACHE STRING "Maximum simultaneous Globe views")
if(NOT ROCKY_MAX_NUMBER_OF_VIEWS MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "ROCKY_MAX_NUMBER_OF_VIEWS must be a positive integer")
endif()
set(ROCKY_USE_UTF8_FILENAMES_ON_WINDOWS ON)
foreach(FEATURE GLM PROJ SPDLOG JSON CURL GDAL IMGUI VSG VSGXCHANGE ENTT)
    set(ROCKY_HAS_${FEATURE} ON)
endforeach()
if(EXISTS "${GDAL_INCLUDE_DIR}/ogr_geocoding.h")
    set(ROCKY_HAS_GEOCODER ON)
endif()
set(GLOBE_INCLUDE "${PROJECT_SOURCE_DIR}/include/geostudio/native/globe")
configure_file("${PROJECT_SOURCE_DIR}/resources/globe/shaders/rocky.defines.h.glsl" "${GEOSTUDIO_GENERATED_INCLUDE}/geostudio/native/globe/vsg/shaders/rocky.defines.h.glsl" COPYONLY)
configure_file(Version.h.in "${GEOSTUDIO_GENERATED_INCLUDE}/geostudio/native/globe/Version.h")
file(GLOB SOURCES CONFIGURE_DEPENDS *.cpp tinyxml/*.cpp contrib/*.cpp ecs/*.cpp vsg/*.cpp vsg/terrain/*.cpp vsg/ecs/*.cpp vsg/imgui/*.cpp)
list(REMOVE_ITEM SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/vsg/ecs/SlugSystem.cpp")
add_library(geostudio-globe STATIC ${SOURCES})
set_target_properties(geostudio-globe PROPERTIES EXPORT_NAME Globe POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(geostudio-globe PRIVATE ROCKY_BUILDING_SDK PUBLIC ROCKY_STATIC)
target_compile_options(geostudio-globe PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/permissive-> $<$<CXX_COMPILER_ID:MSVC>:/we4834> $<$<CXX_COMPILER_ID:MSVC>:/bigobj>)
target_include_directories(geostudio-globe PUBLIC
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
    $<BUILD_INTERFACE:${GEOSTUDIO_GENERATED_INCLUDE}>
    $<INSTALL_INTERFACE:include>
    PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(geostudio-globe PUBLIC glm::glm spdlog::spdlog EnTT::EnTT vsg::vsg
    PRIVATE PROJ::proj GDAL::GDAL CURL::libcurl nlohmann_json::nlohmann_json imgui::imgui geostudio-io)
if(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(geostudio-globe PUBLIC Threads::Threads)
endif()
install(TARGETS geostudio-globe EXPORT GeoStudioNativeTargets ARCHIVE DESTINATION lib)
