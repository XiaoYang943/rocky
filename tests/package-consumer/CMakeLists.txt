cmake_minimum_required(VERSION 3.22)
project(GeoStudioNativeConsumer LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MsvcIncludes.cmake")
set(CMAKE_DISABLE_FIND_PACKAGE_imgui ON)
find_package(GeoStudioNative 0.3 CONFIG REQUIRED)
if(TARGET imgui::imgui)
    message(FATAL_ERROR "SDK package unexpectedly creates an ImGui dependency")
endif()
add_executable(native-sdk-consumer main.cpp)
target_compile_features(native-sdk-consumer PRIVATE cxx_std_17)
target_link_libraries(native-sdk-consumer PRIVATE GeoStudio::Native)
add_executable(native-sdk-core-consumer core.cpp)
target_compile_features(native-sdk-core-consumer PRIVATE cxx_std_17)
target_link_libraries(native-sdk-core-consumer PRIVATE GeoStudio::Data)
