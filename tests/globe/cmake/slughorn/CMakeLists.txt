cmake_minimum_required(VERSION 3.21)
project(rocky_slughorn_smoke LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(GEOSTUDIO_NATIVE_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(_adapter "${GEOSTUDIO_NATIVE_SOURCE_DIR}/src/render/entity/slug")
set(_adapter_include "${CMAKE_CURRENT_BINARY_DIR}/include")
configure_file("${_adapter}/SlugAdapter.h" "${_adapter_include}/geostudio/native/render/entity/slug/SlugAdapter.h" COPYONLY)
configure_file("${GEOSTUDIO_NATIVE_SOURCE_DIR}/src/render/entity/SlugCoverage.h" "${_adapter_include}/geostudio/native/render/entity/SlugCoverage.h" COPYONLY)

# The consumer mode tests the installed static target's transitive link-only SDK
# dependency, rather than accidentally relying on build-tree target visibility.
if(SLUGHORN_TEST_INSTALLED_CONSUMER)
    find_package(slughorn_smoke CONFIG REQUIRED)
else()
    list(APPEND CMAKE_MODULE_PATH "${GEOSTUDIO_NATIVE_SOURCE_DIR}/cmake")
    include(ResolveSlughorn)
    rocky_resolve_slughorn()
    add_library(slughorn_smoke_adapter STATIC "${_adapter}/SlugAdapter.cpp")
    target_include_directories(slughorn_smoke_adapter PRIVATE "${_adapter_include}")
    set_target_properties(slughorn_smoke_adapter PROPERTIES CXX_STANDARD 20)
    target_link_libraries(slughorn_smoke_adapter PRIVATE slughorn::slughorn)
    install(TARGETS slughorn_smoke_adapter EXPORT slughornSmokeTargets ARCHIVE DESTINATION lib)
    install(EXPORT slughornSmokeTargets DESTINATION lib/cmake/slughorn_smoke)
    install(FILES "${CMAKE_CURRENT_LIST_DIR}/slughorn_smoke-config.cmake" DESTINATION lib/cmake/slughorn_smoke)
endif()

add_executable(slughorn_smoke smoke.cpp)
target_include_directories(slughorn_smoke PRIVATE "${_adapter_include}")
target_link_libraries(slughorn_smoke PRIVATE slughorn_smoke_adapter)
add_test(NAME slughorn_smoke COMMAND slughorn_smoke)
